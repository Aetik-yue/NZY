```dart
  Future<void> next() async {
    if (state.playlist.isEmpty) return;
    final len = state.playlist.length;
    final idx = state.currentIndex < 0 ? 0 : state.currentIndex;
    switch (state.playMode) {
      case PlayMode.repeatOne:
        // 手动点下一首仍切下一首（网易云行为）；播完自动切歌才重播当前
        await selectSong((idx + 1) % len);
      case PlayMode.shuffle:
        if (len <= 1) {
          await selectSong(0);
          return;
        }
        var r = _random.nextInt(len);
        while (r == idx) {
          r = _random.nextInt(len);
        }
        await selectSong(r);
      case PlayMode.sequence:
        await selectSong((idx + 1) % len);
    }
  }
```

6. `_onTrackCompleted` 按模式（单曲循环重播当前）：

```dart
  /// 一首播完自动切下一首；单曲循环模式重播当前；失败已被 selectSong 内部兜底
  Future<void> _onTrackCompleted() async {
    try {
      if (state.playMode == PlayMode.repeatOne &&
          state.currentIndex >= 0) {
        // 单曲循环：seek 回开头并继续播放
        await _backend.seek(Duration.zero);
        await _backend.play();
        state = state.copyWith(position: Duration.zero);
        return;
      }
      await next();
    } catch (e) {
      debugPrint('自动切歌失败: $e');
    }
  }
```

```dart
/// 进度条行（Slider + 时间标签），拖动中仅预览、松手才 seek
class MusicProgressRow extends ConsumerStatefulWidget {
  final MusicPlayerState state;
  const MusicProgressRow({super.key, required this.state});

  @override
  ConsumerState<MusicProgressRow> createState() => _MusicProgressRowState();
}

class _MusicProgressRowState extends ConsumerState<MusicProgressRow> {
  /// 拖动中的预览值（0..1），null = 未拖动
  double? _dragValue;

  @override
  Widget build(BuildContext context) {
    final total = widget.state.duration?.inSeconds ?? 0;
    // 拖动中显示预览位置，否则显示播放位置
    final seconds = _dragValue != null
        ? _dragValue! * total
        : widget.state.position.inSeconds.clamp(0, total > 0 ? total : 0);
    return Padding(
      padding: const EdgeInsets.symmetric(horizontal: 24),
      child: Column(
        children: [
          Slider(
            value: total > 0 ? seconds / total : 0,
            min: 0,
            max: 1,
            activeColor: AppColors.accentRose,
            onChanged: total > 0
                ? (v) => setState(() => _dragValue = v) // 拖动中不 seek
                : null,
            onChangeStart: total > 0
                ? (_) => setState(() =>
                    _dragValue = (widget.state.position.inSeconds / total)
                        .clamp(0.0, 1.0))
                : null,
            onChangeEnd: total > 0
                ? (v) {
                    setState(() => _dragValue = null);
                    ref.read(musicPlayerProvider.notifier).seekTo(
                        Duration(seconds: (v * total).round()));
                  }
                : null,
          ),
          Row(
            mainAxisAlignment: MainAxisAlignment.spaceBetween,
            children: [
              Text(_fmt(Duration(seconds: seconds.round())),
                  style: AppTextStyles.caption
                      .copyWith(color: AppColors.textOnPink)),
              Text(_fmt(widget.state.duration ?? Duration.zero),
                  style: AppTextStyles.caption
                      .copyWith(color: AppColors.textOnPink)),
            ],
          ),
        ],
      ),
    );
  }

  String _fmt(Duration d) {
    final m = d.inMinutes.remainder(60).toString().padLeft(2, '0');
    final s = d.inSeconds.remainder(60).toString().padLeft(2, '0');
    return '$m:$s';
  }
}
```

（`_fmt` 方法原样保留；`widget.state` 替代原 `state` 参数。）

        .animate(CurvedAnimation(parent: _ctrl, curve: Curves.easeInOut));
    _b = Tween(begin: 1.0, end: 0.3)
        .animate(CurvedAnimation(parent: _ctrl, curve: Curves.easeInOut));
    _c = Tween(begin: 0.6, end: 0.9)
        .animate(CurvedAnimation(parent: _ctrl, curve: Curves.easeInOut));
  }

  @override
  void didUpdateWidget(_EqualizerBars oldWidget) {
    super.didUpdateWidget(oldWidget);
    if (widget.playing && !_ctrl.isAnimating) {
      _ctrl.repeat(reverse: true);
    } else if (!widget.playing && _ctrl.isAnimating) {
      _ctrl.stop();
      _ctrl.value = 0.5; // 静止在中间高度
    }
  }

  @override
  void dispose() {
    _ctrl.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return AnimatedBuilder(
      animation: _ctrl,
      builder: (context, _) {
        final barHeight = 14.0;
        return Row(
          mainAxisSize: MainAxisSize.min,
          crossAxisAlignment: CrossAxisAlignment.center,
          children: [
            _bar(_a.value * barHeight),
            const SizedBox(width: 2),
            _bar(_b.value * barHeight),
            const SizedBox(width: 2),
            _bar(_c.value * barHeight),
          ],
        );
      },
    );
  }

  Widget _bar(double height) {
    return Container(
      width: 3,
      height: height.clamp(4, 14),
      decoration: BoxDecoration(
        color: widget.playing
            ? AppColors.accentRose
            : AppColors.textTertiary,
        borderRadius: BorderRadius.circular(AppColors.radiusXs),
      ),
    );
  }
}
